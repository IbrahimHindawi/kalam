cmake_minimum_required(VERSION 3.24)
project(kalam_native LANGUAGES C CXX)
include(FetchContent)
if(NOT EXISTS "${KALAM_RUST_LIB}")
  message(FATAL_ERROR "Set KALAM_RUST_LIB to the Rust static library built for this target")
endif()
if(SDL3_DIR AND NOT DEFINED FETCHCONTENT_SOURCE_DIR_SDL3)
  find_package(SDL3 CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${SDL3_DIR}")
endif()
if(NOT TARGET SDL3::SDL3)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  if(ANDROID)
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  elseif(WIN32)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  endif()
  FetchContent_Declare(SDL3
    URL https://codeload.github.com/libsdl-org/SDL/zip/877399b2b2cf21e67554ed9046410f268ce1d1b2
    URL_HASH SHA256=6ad9ee42f6a08bc5cbc113a43b71ab59af0414625948287c7a14d8d4f631bc2b
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3)
endif()
set(BGFX_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BGFX_INSTALL OFF CACHE BOOL "" FORCE)
set(BGFX_CONFIG_MULTITHREADED OFF CACHE BOOL "" FORCE)
set(BGFX_CONFIG_VIDEO OFF CACHE BOOL "" FORCE)
# Wrapper revision pins bgfx 0762725e, bimg 9bdb5805 and bx f8508dc0.
if(bgfx_DIR)
  find_package(bgfx CONFIG REQUIRED)
  set(kh_bgfx bgfx::bgfx)
else()
  FetchContent_Declare(bgfx_cmake GIT_REPOSITORY https://github.com/bkaradzic/bgfx.cmake.git GIT_TAG 82f885c710b98a4b4f4df40cc234a3cb702f4846)
  FetchContent_MakeAvailable(bgfx_cmake)
  set(kh_bgfx bgfx)
endif()
if(ANDROID)
  set(host main)
  add_library(main SHARED host.c render.c ../android/surface.c)
elseif(WIN32)
  set(host kalam)
  add_executable(kalam host.c render.c)
else()
  message(FATAL_ERROR "This renderer currently qualifies Windows and Android")
endif()
target_compile_features(${host} PRIVATE c_std_11)
set(KALAM_GAME_NAME "" CACHE STRING "Name of the game statically linked into the host")
if(KALAM_GAME_NAME)
  target_compile_definitions(${host} PRIVATE KALAM_GAME_APPLICATION)
  set_target_properties(${host} PROPERTIES OUTPUT_NAME "${KALAM_GAME_NAME}")
endif()
target_link_libraries(${host} PRIVATE SDL3::SDL3 ${kh_bgfx} "${KALAM_RUST_LIB}")
if(WIN32)
  target_link_libraries(${host} PRIVATE ws2_32 userenv bcrypt ntdll synchronization)
else()
  target_link_libraries(${host} PRIVATE android log dl m)
endif()
set_target_properties(${host} PROPERTIES LINKER_LANGUAGE CXX)
option(KALAM_EDITOR "Build the SDL3/bgfx Windows editor" OFF)
if(KALAM_EDITOR)
  if(NOT WIN32 OR ANDROID)
    message(FATAL_ERROR "The editor currently hosts a Windows viewport")
  endif()
  FetchContent_Declare(kh_imgui
    URL https://codeload.github.com/ocornut/imgui/zip/refs/tags/v1.91.9b-docking
    URL_HASH SHA256=ca3471ed0c472b210fc37242e0b6338f37574f77f8bc1303befc040f0c22f06b
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(kh_imgui)
  target_sources(${host} PRIVATE imgui_bridge.cpp
    ${kh_imgui_SOURCE_DIR}/imgui.cpp
    ${kh_imgui_SOURCE_DIR}/imgui_draw.cpp
    ${kh_imgui_SOURCE_DIR}/imgui_tables.cpp
    ${kh_imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${kh_imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp)
  target_include_directories(${host} PRIVATE ${kh_imgui_SOURCE_DIR})
  target_compile_definitions(${host} PRIVATE KALAM_NATIVE_EDITOR IMGUI_USE_WCHAR32)
  target_compile_features(${host} PRIVATE cxx_std_20)
endif()
